% can tell that it happened.
%
%   emitter        training/holdout_eval.py --writeup <model dirs> --verify-against <run json>
%                    |  runs ONE inference pass over the frozen reporting set
%                    v
%   generated/     results_macros.tex  results_table.tex  sweep_table.tex
%                  confusion_table.tex  gate_derivation.tex
%                  finding_gate_degeneracy.tex  limitations.tex
%                    |  every table stamped with the run fingerprint
%                    v
%   results.tex    prose, quoting ONLY macros, \input-ing the tables
%                    |
%                    v
%   main.tex       this file
%
%   Build:  cd kinyamed/ml_model/paper && latexmk -pdf -halt-on-error main.tex
%   Check:  python paper/check_tex.py paper/main.tex      (no engine required)
%

% ---------------------------------------------------------------------------
% ACL TWO-COLUMN STYLE, VENDORED.
%
% acl.sty and acl_natbib.bst sit in this directory rather than being taken from
% whatever TeX Live the compiling machine happens to have. A style file that
% differs between two machines changes the page count, and the page count is a
% submission constraint.
%
% Source: https://github.com/acl-org/acl-style-files (repository root).
%
% acl.sty LOADS geometry, natbib AND hyperref ITSELF, and calls \twocolumn. Do
% not load any of those here: a second \usepackage with different options is an
% option clash, and geometry loaded twice silently wins on the last call.
%
% The previous article-class preamble is kept verbatim at main_article.tex.bak
% so this switch can be reverted in one move if it will not compile.
% ---------------------------------------------------------------------------
\usepackage[preprint]{acl}

% T1 SO THAT UNDERSCORES SURVIVE. Under the default OT1 encoding an underscore
% in typewriter text is not reliably available as a glyph, and the compiled v2
% PDF dropped it from every file path it printed: a paper about reproducibility
% was naming scripts that do not exist under the names it gave. T1 gives the
% character outright.
\usepackage[T1]{fontenc}
% Type 1 fonts. Without these, [T1]{fontenc} makes pdflatex fall back to
% bitmapped EC fonts, which embed as Type 3: blurry on screen, poor in print,
% and flagged by arXiv. mathptmx is Times for text and maths; the ACL
% templates use Times for the same reason.
\usepackage{mathptmx}
\usepackage{helvet}
\usepackage{courier}
% \path{...} for file paths and identifiers. It takes the characters verbatim,
% so a path is written the way it appears on disk with nothing to escape, and
% no future path can lose an underscore to a forgotten backslash.
\usepackage{url}
\usepackage{booktabs}   % \toprule \midrule \bottomrule, used by every table
\usepackage{amsmath}    % \[ \] and \frac, used by the gate derivation
\usepackage{graphicx}
\usepackage{pgfplots}   % the seeds-to-rows figure
\pgfplotsset{compat=1.18}
\usepackage{colortbl}   % \cellcolor for the outcome column
\usepackage{xcolor}
\usepackage{tabularx}   % a shared measure for the two halves of tab:example

% tcolorbox for the takeaway boxes, loaded AFTER acl.sty so nothing it pulls in
% is re-requested with different options.
%
% NO OPTION CLASH ON xcolor: acl.sty line 33 loads xcolor with no options, the
% line above does the same, and tcolorbox requires it with none. A clash needs
% two loads with DIFFERENT options; three optionless loads are one load.
% tcolorbox also pulls in pgf/tikz, which pgfplots has already loaded above.
%
% skins     -- needed for `enhanced`, without which `borderline west` does nothing.
% breakable -- a takeaway landing near the foot of a column splits across the
%              break instead of overflowing the column. In a two-column layout
%              with this many floats that is not a rare case.
\usepackage{tcolorbox}
\tcbuselibrary{skins,breakable}

% \raggedbottom: let a short page end short rather than stretching the vertical
% space between its blocks to reach the bottom margin. With full-width floats on
% a two-column page the default \flushbottom opens visible gaps between
% paragraphs. This trades a ragged page bottom for even interline spacing, which
% is the right trade in a document with this many table* floats.
%
% NOT VERIFIED HERE. There is no TeX engine on this machine, so whether the gap
% this is meant to close is actually gone can only be seen in a compile.
\raggedbottom

% Takeaway box. Short bold heading, two or three lines, after each results
% subsection.
%
% Was \fbox with a full black rule until 2026-09-19, on the stated grounds that
% tcolorbox was one more package that could fail on another machine. That was an
% assumption, not a measurement; tcolorbox and xcolor are both in TeX Live and
% the document now compiles here, so the risk is testable and was tested.
%
% A left accent bar and a light tint rather than a full frame: this is a
% clinical-safety paper and a saturated box reads as a warning it is not. Both
% colours are desaturated slate, defined ONCE below and never written inline, so
% restyling is one edit rather than three that can disagree.
\definecolor{takeawayaccent}{HTML}{44607A}  % desaturated slate blue
\definecolor{takeawaytint}{HTML}{F3F6F9}    % the same hue at very low saturation

% Numbered by counter, NOT by hand. The three headings read "Takeaway 1:",
% "Takeaway 2:" and "Takeaway 3:" as literal text until 2026-09-19, so inserting
% one in the middle would have renumbered nothing and said so in silence.
\newcounter{takeaway}

% Styled after the boxed takeaways in Yu et al. (2026): a rounded frame with a
% filled title bar, rather than a rule in the margin. The title bar carries the
% counter, so a box cannot be mistaken for a block quotation, and the frame
% closes the box at the foot, which a one-sided rule does not.
%
% breakable is load-bearing: these sit after results subsections, which is where
% column breaks fall in two columns. Without it a box at a break overflows the
% column rather than splitting across it.
\newtcolorbox{takeawaybox}[1]{%
  enhanced, breakable,
  colback=takeawaytint,
  colframe=takeawayaccent,
  colbacktitle=takeawayaccent,
  coltitle=white,
  boxrule=0.6pt,
  rounded corners,
  arc=2.5pt,
  title={\ensuremath{\triangleright}~\textbf{Takeaway \thetakeaway: #1}},
  fonttitle=\small,
  % The title is NOT attached as a floating boxed title: an attached title does
  % not wrap, and Takeaway 3's heading ran to x=596 on a 595pt page. A plain
  % title fills the box width and breaks across lines like any other text.
  toptitle=1.5pt, bottomtitle=1.5pt,
  left=7pt, right=7pt, top=5pt, bottom=6pt,
  before skip=0.9em, after skip=0.9em,
  fontupper=\small,
}

% \triangleright, not \blacktriangleright: the filled form lives in amssymb,
% which this document does not load. This one is a standard math symbol, so it
% sets in whatever maths font the preamble selects -- Computer Modern when this
% was written, Times since mathptmx was added on 2026-09-19. Phrased so it does
% not have to be revisited the next time the font changes.
% The `~` after the marker is deliberate and so is the absence of a backslash
% at the end of any line in this macro: `\` immediately before a newline is the
% null control sequence, not a control space, and it is an error rather than the
% thin gap it looks like.
\newcommand{\takeaway}[2]{%
  \par
  \refstepcounter{takeaway}%
  \begin{takeawaybox}{#1}%
    #2%
  \end{takeawaybox}%
  \par
}

% APPENDIX HEADINGS READ "Appendix H: Title", NOT "H Title".
%
% A bare letter before a title reads as a stray character to anyone not
% expecting LaTeX's appendix numbering. This is set once, here, rather than by
% editing nine \section titles by hand.
%
% \@startsection calls \@seccntformat to typeset the number, so patching it
% reaches every heading at every level. The prefix and suffix are looked up by
% level name: \sectionprefix for a section, \subsectionprefix for a
% subsection. An undefined one expands to \relax and prints nothing, so main
% body sections are untouched and only the levels we name gain a prefix.
%
% \thesection IS DELIBERATELY NOT CHANGED. Redefining it to "Appendix H" would
% make every "Appendix~\ref{app:run}" in the prose read "Appendix Appendix F".
% This affects the heading only; \ref still yields the bare letter.
\makeatletter
\renewcommand{\@seccntformat}[1]{%
  \csname #1prefix\endcsname
  \csname the#1\endcsname
  \csname #1suffix\endcsname
  \quad
}
\makeatother

% ---------------------------------------------------------------------------
% FAIL LOUDLY ON A MISSING NUMBER.
%
% An undefined control sequence is already a LaTeX error, but it reports as
% "Undefined control sequence \ResultMacroFOne" with no indication of where the
% macro was supposed to come from. Someone seeing that at 2am will define it by
% hand, and then the paper carries a number nothing computed.
%
% So every figure the prose depends on is asserted here by name, with an error
% message that says which command regenerates it. A missing macro stops the
% build with an instruction rather than a puzzle.
% ---------------------------------------------------------------------------
\makeatletter
\newcommand{\RequireGenerated}[1]{%
  \@ifundefined{#1}{%
    \GenericError{}{%
      PROVENANCE ERROR: macro #1 is not defined%
    }{}{%
      This macro must come from generated/results_macros.tex, which is written
      by: python training/holdout_eval.py --writeup <model dirs> --verify-against
      <run json>. Re-run the emitter. Do NOT define it by hand -- a number
      typed here is a number nothing computed.%
    }%
  }{}%
}
\makeatother
