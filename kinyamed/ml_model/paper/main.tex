% BUILD DRIVER FOR THE RESULTS SECTION.
%
% The document is assembled by the same pipeline that computes the numbers.
% That is the point: a manuscript written elsewhere and filled in by hand
% breaks the provenance chain at the first copy-paste, and nothing downstream
% can tell that it happened.
%
%   emitter        training/holdout_eval.py --writeup <model dirs> --verify-against <run json>
%                    |  runs ONE inference pass over the frozen reporting set
%                    v
%   generated/     results_macros.tex  results_table.tex  sweep_table.tex
%                  confusion_table.tex  gate_derivation.tex
%                  finding_gate_degeneracy.tex  limitations.tex
%                    |  every table stamped with the run fingerprint
%                    v
%   results.tex    prose, quoting ONLY macros, \input-ing the tables
%                    |
%                    v
%   main.tex       this file
%
%   Build:  cd kinyamed/ml_model/paper && latexmk -pdf -halt-on-error main.tex
%   Check:  python paper/check_tex.py paper/main.tex      (no engine required)
%
% The prose sections live in sections/ and are empty stubs. The results
% section and the limitations statement are NOT stubs: they are assembled from
% generated fragments and must not be rewritten by hand.

% hyperref's defaults draw a coloured border around every link: red for an
% internal reference, green for a citation. They render as boxes in the PDF and
% are printed by some viewers. hidelinks removes the border and the colour while
% leaving the link live, which is what a preprint wants.
\hypersetup{hidelinks}

% ---------------------------------------------------------------------------
% FAIL LOUDLY ON A MISSING NUMBER.
%
% An undefined control sequence is already a LaTeX error, but it reports as
% "Undefined control sequence \ResultMacroFOne" with no indication of where the
% macro was supposed to come from. Someone seeing that at 2am will define it by
% hand, and then the paper carries a number nothing computed.
%
% So every figure the prose depends on is asserted here by name, with an error
% message that says which command regenerates it. A missing macro stops the
% build with an instruction rather than a puzzle.
% ---------------------------------------------------------------------------
\makeatletter
\newcommand{\RequireGenerated}[1]{%
  \@ifundefined{#1}{%
    \GenericError{}{%
      PROVENANCE ERROR: macro #1 is not defined%
    }{}{%
      This macro must come from generated/results_macros.tex, which is written
      by: python training/holdout_eval.py --writeup <model dirs> --verify-against
      <run json>. Re-run the emitter. Do NOT define it by hand -- a number
      typed here is a number nothing computed.%
    }%
  }{}%
}
\makeatother

% The macros the prose actually quotes. An entry here that the emitter stops
% producing is caught at build time.
%
% This list held 21 entries until 2026-09-17, of which 20 named macros nothing
% in the document printed. A guard asserting the presence of a number the paper
% does not use cannot fail in any way that matters, and a list of twenty such
% guards reads as provenance while providing none.
%
% The numbers themselves did not lose their provenance when the prose stopped
% quoting macros: every one of them reaches the paper inside a generated .tex
% file that this document \input s, written by the same emitter in the same
% pass. See tests/test_render_completeness.py, which checks the four structural
% counts that results.tex also states in prose against the emitter's values --
% the one place where a regenerated figure could silently disagree with the
% text around it.
\input{generated/results_macros}

\RequireGenerated{ResultFingerprint}

% claim, and this project has already been bitten by a generated file that
% quietly outlived its truth.
