% Takeaway box. Short bold heading, two or three lines, after each results
% subsection. Deliberately plain: a framed box that needs tcolorbox is one more

\newcommand{\takeaway}[2]{%
  \par\vspace{0.6em}%
  \noindent\fbox{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
    \small\textbf{#1}\ #2}}%
  \par\vspace{0.6em}%
}

% APPENDIX HEADINGS READ "Appendix H: Title", NOT "H Title".
%
% A bare letter before a title reads as a stray character to anyone not
% expecting LaTeX's appendix numbering. This is set once, here, rather than by
% editing nine \section titles by hand.
%
% \@startsection calls \@seccntformat to typeset the number, so patching it
% reaches every heading at every level. The prefix and suffix are looked up by
% level name: \sectionprefix for a section, \subsectionprefix for a
% subsection. An undefined one expands to \relax and prints nothing, so main
% body sections are untouched and only the levels we name gain a prefix.
%
% \thesection IS DELIBERATELY NOT CHANGED. Redefining it to "Appendix H" would
% make every "Appendix~\ref{app:run}" in the prose read "Appendix Appendix F".
% This affects the heading only; \ref still yields the bare letter.
\makeatletter
\renewcommand{\@seccntformat}[1]{%
  \csname #1prefix\endcsname
  \csname the#1\endcsname
  \csname #1suffix\endcsname
  \quad
}
\makeatother

% ---------------------------------------------------------------------------
% FAIL LOUDLY ON A MISSING NUMBER.
%
% An undefined control sequence is already a LaTeX error, but it reports as
% "Undefined control sequence \ResultMacroFOne" with no indication of where the
% macro was supposed to come from. Someone seeing that at 2am will define it by
% hand, and then the paper carries a number nothing computed.
%
% So every figure the prose depends on is asserted here by name, with an error
% message that says which command regenerates it. A missing macro stops the
% build with an instruction rather than a puzzle.
% ---------------------------------------------------------------------------
\makeatletter
\newcommand{\RequireGenerated}[1]{%
  \@ifundefined{#1}{%
    \GenericError{}{%
      PROVENANCE ERROR: macro #1 is not defined%
    }{}{%
      This macro must come from generated/results_macros.tex, which is written
      by: python training/holdout_eval.py --writeup <model dirs> --verify-against
      <run json>. Re-run the emitter. Do NOT define it by hand -- a number
      typed here is a number nothing computed.%
    }%
  }{}%
}
\makeatother
